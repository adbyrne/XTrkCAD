# build unit tests for the xtrkcad library
#

# The include directories nearly every test needs: app/bin, its include/
# and the wlib headers.
set(_xtrkcad_ut_includes
    ${CMAKE_CURRENT_SOURCE_DIR}/..
    ${CMAKE_CURRENT_SOURCE_DIR}/../include
    ${CMAKE_SOURCE_DIR}/app/wlib/include
)

# xtrkcad_add_unit_test(<target> <test> [MATH] [SOURCES <file>...]
#                       [INCLUDES <dir>...] [LIBS <library>...])
#
# One unit test: the executable <target> from <target>.c and SOURCES, with
# the INCLUDES directories, linked with LIBS and CMocka (and libm where there
# is one, with MATH), the ctest test <test>, and the UnitTests folder. A test
# that needs more (dependencies, link options, properties) adds them after
# the call.
function(xtrkcad_add_unit_test _target _test)
    cmake_parse_arguments(PARSE_ARGV 2 _ut "MATH" "" "SOURCES;INCLUDES;LIBS")
    add_executable(${_target} ${_target}.c ${_ut_SOURCES})
    if(_ut_INCLUDES)
        target_include_directories(${_target} PRIVATE ${_ut_INCLUDES})
    endif()
    target_link_libraries(${_target} PRIVATE ${_ut_LIBS} CMocka::cmocka)
    if(_ut_MATH)
        target_link_libraries(${_target} PRIVATE "$<${XTRKCAD_IF_UNIX}:m>")
    endif()
    add_test(${_test} ${_target})
    set_target_properties(${_target} PROPERTIES FOLDER UnitTests)
endfunction()

# misc.c #includes common-ui.h -> messages.h, generated by the msgfiles
# target (see app/help/CMakeLists.txt) -- same dependency app/bin's own
# xtrkcad-lib target declares.
add_dependencies(getargstest msgfiles)

# misc.c is NOT isolated the way getargs.c is -- it contains wMain(), the
# whole application's entry point, transitively referencing ~100 other
# symbols (menu/toolbar/undo/pref/audio/splash-screen init, etc.) that have
# nothing to do with ConvertToEscapedText()/ConvertFromEscapedText(), the
# only two functions this test actually needs from it.
#
# Originally tried relying purely on --gc-sections/-ffunction-sections
# (with ENABLE_EXPORTS forced off, since PlatformSettings.cmake's global
# CMAKE_ENABLE_EXPORTS TRUE otherwise adds --export-dynamic/-rdynamic and
# keeps every global symbol "live") to let the linker discard wMain()'s
# whole unreachable call graph -- this worked on Linux and macOS, but not
# on Windows/MSYS2: confirmed via the actual CI job log that MinGW's ld
# still pulled in wMain() and ~90 of its transitive dependencies as
# unresolved externals despite the same flags, ENABLE_EXPORTS FALSE
# included. Root cause not fully pinned down (Windows PE/COFF's
# --export-all-symbols handling for EXE targets appears to keep global
# symbols reachable for the link's purposes in a way ELF/Mach-O don't), so
# rather than chase a third platform-specific linker workaround, wMain()'s
# body is now excluded from compilation entirely for this target via
# XTRKCAD_TESTBUILD_NO_WMAIN (misc.c's own #ifndef guard around it) --
# deterministic on every platform, since the symbol and its call graph
# never exist in this target's object file at all. The gc-sections/
# dead_strip flags below are kept as a secondary, no-longer-load-bearing
# cleanup for any other genuinely-unused code in the two linked files.
#
# Side effect: any misc.c static that's only reachable from wMain() (e.g.
# SetAccelKeys/OfferCheckpoint/AccelKeyDispatch/NextWindow, SF #796) shows up
# as "defined but not used" in this target's -Wall -Wextra build once its
# only caller is excluded, and must be wrapped in the same #ifndef guard.
# SF #797 tracks extracting ConvertToEscapedText()/ConvertFromEscapedText()
# into their own file so this target stops compiling all of misc.c and this
# whole class of warning goes away permanently.
set_target_properties(getargstest PROPERTIES ENABLE_EXPORTS FALSE)
target_compile_definitions(getargstest PRIVATE XTRKCAD_TESTBUILD_NO_WMAIN)
target_compile_options(getargstest PRIVATE -ffunction-sections -fdata-sections)
# Apple's ld64 doesn't understand GNU's --gc-sections syntax at all
# ("ld: unknown options: --gc-sections", confirmed on the macOS CI runner)
# -- -dead_strip is its equivalent flag for the same per-symbol dead-code
# elimination.
target_link_options(getargstest
	PRIVATE
	"$<${XTRKCAD_IF_MACOS}:-Wl,-dead_strip>"
	"$<$<NOT:$<PLATFORM_ID:Windows,Darwin>>:-Wl,--gc-sections>"
)
